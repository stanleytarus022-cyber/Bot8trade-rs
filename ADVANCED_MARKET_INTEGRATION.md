bool,\n}\n\nimpl DynamicPositionSizer {\n    pub fn calculate_position_size(\n        &self,\n        entry_price: f64,\n        stop_loss: f64,\n        atr: f64,\n    ) -> f64 {\n        let risk_amount = self.account_size * (self.max_risk_percent / 100.0);\n        let price_risk = (entry_price - stop_loss).abs();\n        \n        // Adjust for volatility if enabled\n        let Formula: f* = (bp - q) / b\n        // where: b = ratio of win to loss\n        //        p = probability of win\n        //        q = probability of loss (1-p)\n        \n        let b = avg_win / avg_loss;\n        let p = win_rate;\n        let q = loss_rate;\n        \n        ((b * p - q) / b).max(0.0).min(0.25) // Cap at 25%\n    }\n}\n```\n\n### 5.2 Correlation-Based Risk
